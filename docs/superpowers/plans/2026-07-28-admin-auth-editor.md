# Admin auth + article editor — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the public, unauthenticated `/edit/$slug` route with a `/admin` area gated by WorkOS AuthKit (Google login) plus an email allowlist, reusing the existing raw-HTML article form.

**Architecture:** TanStack Start server functions wrap `useSession` (built into `@tanstack/react-start/server`) for a stateless, sealed httpOnly cookie session holding `{ userId, email }`. A single shared helper (`src/lib/admin-auth.ts`) exposes a `checkAdminAuth` server function that every protected route's `beforeLoad` calls; unauthenticated visitors are redirected to `/admin/login`, authenticated-but-not-allowlisted visitors get a "No autorizado" render (no redirect loop). WorkOS's Node SDK builds the Google OAuth URL and exchanges the callback code for a user. Routes follow this repo's existing flat dot-file convention (e.g. `article.$slug.tsx` → `/article/$slug`). Exception: once `admin.tsx` (the `/admin` list route) exists, any sibling starting with `admin.` becomes a real nested child of it in TanStack Router's route tree — `admin.tsx`'s component has no `<Outlet />`, so a genuinely nested child's component would never render. `/admin/login`, `/admin/logout`, `/admin/callback`, `/admin/new`, and `/admin/edit/$slug` all use the `admin_.` escape prefix (e.g. `admin_.new.tsx`) instead, which keeps their URL paths unchanged but re-parents them to the router root. See Task 6 and Task 7 for how this was discovered and fixed.

**Tech Stack:** TanStack Start 1.168 / TanStack Router (file-based, flat dot convention), Prisma 6 + Postgres, `@workos-inc/node` 10.8 (already installed via `bun add @workos-inc/node`), Bun as package manager (`bun.lock`).

## Global Constraints

- No Prisma schema changes — reuse `Article` model as-is (spec non-goal).
- No delete-article flow (spec non-goal, YAGNI).
- No DB-backed session table — session is a stateless sealed cookie via TanStack Start's built-in `useSession` (spec non-goal: "No DB-backed session table").
- Allowlist (`ADMIN_EMAILS`) is a comma-separated env var, no self-service UI (spec non-goal).
- `src/routes/edit.$slug.tsx` is deleted entirely; `/edit/$slug` must no longer exist.
- This repo has **no test framework installed** (no vitest/jest, no `*.test.*` files, no test script in `package.json`). Per "follow established patterns," this plan does **not** introduce one — "tests" below are manual verification steps (dev server + curl / browser), not automated test files. Do not add a test framework as part of this plan.
- Package manager is **Bun** (`bun.lock` present, no `package-lock.json`/`pnpm-lock.yaml`). Use `bun add` / `bun run` in all steps.
- `src/routeTree.gen.ts` is gitignored and auto-generated by the `tanstackStart` Vite plugin on `bun run dev` / `bun run build`. New route files won't be importable/typed until the dev server (or a build) has run at least once after adding them.
- Follow existing code style: server functions via `createServerFn`, Prisma calls directly in the handler, Tailwind utility classes matching the `--color-wiki-*` custom properties already used in `__root.tsx` / `edit.$slug.tsx`, Spanish-language UI copy (all existing UI text is in Spanish — "Guardando…", "Buscar", etc.).

```typescript
// src/lib/admin-auth.ts
import { redirect } from "@tanstack/react-router";
import { createServerFn } from "@tanstack/react-start";
import { useSession } from "@tanstack/react-start/server";

interface AdminSessionData {
  userId: string;
  email: string;
}

function getAdminSession() {
  return useSession<AdminSessionData>({
    password: process.env.WORKOS_COOKIE_PASSWORD!,
    name: "admin_session",
    cookie: {
      httpOnly: true,
      secure: process.env.NODE_ENV === "production",
      sameSite: "lax",
      path: "/",
    },
  });
}

export async function setAdminSession(userId: string, email: string) {
  const session = await getAdminSession();
  await session.update({ userId, email });
}

export async function clearAdminSession() {
  const session = await getAdminSession();
  await session.clear();
}

export type AdminAuthResult =
  | { status: "unauthenticated" }
  | { status: "unauthorized"; email: string }
  | { status: "ok"; email: string };

export const checkAdminAuth = createServerFn({ method: "GET" }).handler(
  async (): Promise<AdminAuthResult> => {
    const session = await getAdminSession();
    const email = session.data.email;
    if (!email) return { status: "unauthenticated" };

    const allowlist = (process.env.ADMIN_EMAILS ?? "")
      .split(",")
      .map((e) => e.trim().toLowerCase())
      .filter(Boolean);

    if (!allowlist.includes(email.toLowerCase())) {
      return { status: "unauthorized", email };
    }
    return { status: "ok", email };
  },
);

export async function requireAdmin(
  redirectTo: string,
): Promise<AdminAuthResult> {
  const result = await checkAdminAuth();
  if (result.status === "unauthenticated") {
    throw redirect({ to: "/admin/login", search: { redirect: redirectTo } });
  }
  return result;
}

export function NotAuthorized({ email }: { email: string }) {
  return (
    <>
      <h1>No autorizado</h1>
      <p>
        La cuenta <code>{email}</code> inició sesión correctamente pero no
        está en la lista de administradores.
      </p>
      <p>
        <a
          href="/admin/logout"
          className="text-[--color-wiki-link] hover:underline"
        >
          Cerrar sesión
        </a>
      </p>
    </>
  );
}
```

```typescript
// src/routes/admin.callback.tsx
import { createFileRoute, redirect } from "@tanstack/react-router";
import { createServerFn } from "@tanstack/react-start";
import { setAdminSession } from "~/lib/admin-auth";
import { workos } from "~/lib/workos";

const exchangeCode = createServerFn({ method: "GET" })
  .inputValidator((input: { code: string; state?: string }) => input)
  .handler(async ({ data }) => {
    const { user } = await workos.userManagement.authenticateWithCode({
      code: data.code,
    });
    await setAdminSession(user.id, user.email);
    const redirectTo =
      data.state && data.state.startsWith("/") ? data.state : "/admin";
    return { redirectTo };
  });

export const Route = createFileRoute("/admin/callback")({
  validateSearch: (search: Record<string, unknown>) => ({
    code: String(search.code ?? ""),
    state: typeof search.state === "string" ? search.state : undefined,
  }),
  beforeLoad: async ({ search }) => {
    const { redirectTo } = await exchangeCode({
      data: { code: search.code, state: search.state },
    });
    throw redirect({ href: redirectTo });
  },
});
```

`redirect({ href })` (rather than `{ to }`) is used because `redirectTo` is a runtime string that may include a query string (e.g. `/admin/new?slug=foo`) and isn't statically known to the router's type system.

```typescript
// src/lib/article-form.tsx
import { useState } from "react";

export interface ArticleFormValues {
  slug: string;
  title: string;
  summary: string;
  contentHtml: string;
}

function slugify(title: string) {
  return title
    .toLowerCase()
    .normalize("NFD")
    .replace(/[̀-ͯ]/g, "")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/(^-|-$)/g, "");
}

export function ArticleForm({
  initial,
  slugEditable,
  onSubmit,
  submitLabel,
}: {
  initial: ArticleFormValues;
  slugEditable: boolean;
  onSubmit: (values: ArticleFormValues) => Promise<void>;
  submitLabel: string;
}) {
  const [slug, setSlug] = useState(initial.slug);
  const [slugTouched, setSlugTouched] = useState(!slugEditable || initial.slug !== "");
  const [title, setTitle] = useState(initial.title);
  const [summary, setSummary] = useState(initial.summary);
  const [contentHtml, setContentHtml] = useState(initial.contentHtml);
  const [saving, setSaving] = useState(false);

  return (
    <form
      className="space-y-3"
      onSubmit={async (e) => {
        e.preventDefault();
        setSaving(true);
        await onSubmit({ slug, title, summary, contentHtml });
        setSaving(false);
      }}
    >
      {slugEditable && (
        <label className="block">
          <span className="text-sm font-semibold">Slug</span>
          <input
            value={slug}
            onChange={(e) => {
              setSlug(e.target.value);
              setSlugTouched(true);
            }}
            required
            className="block w-full border border-[--color-wiki-border] p-1 bg-white font-mono text-sm"
          />
        </label>
      )}
      <label className="block">
        <span className="text-sm font-semibold">Título</span>
        <input
          value={title}
          onChange={(e) => {
            const value = e.target.value;
            setTitle(value);
            if (slugEditable && !slugTouched) setSlug(slugify(value));
          }}
          required
          className="block w-full border border-[--color-wiki-border] p-1 bg-white"
        />
      </label>
      <label className="block">
        <span className="text-sm font-semibold">Resumen</span>
        <input
          value={summary}
          onChange={(e) => setSummary(e.target.value)}
          className="block w-full border border-[--color-wiki-border] p-1 bg-white"
        />
      </label>
      <label className="block">
        <span className="text-sm font-semibold">Contenido (HTML)</span>
        <textarea
          value={contentHtml}
          onChange={(e) => setContentHtml(e.target.value)}
          rows={20}
          className="block w-full border border-[--color-wiki-border] p-2 font-mono text-sm bg-white"
        />
      </label>
      <button
        type="submit"
        disabled={saving}
        className="border border-[--color-wiki-border] px-4 py-1 bg-[--color-wiki-sidebar] hover:bg-white disabled:opacity-50"
      >
        {saving ? "Guardando…" : submitLabel}
      </button>
    </form>
  );
}
```

`slugEditable` gates whether the slug field renders and whether title-driven auto-slug applies — `/admin/edit/$slug` passes `slugEditable={false}` (slug is fixed, taken from the URL param), `/admin/new` passes `slugEditable={true}`.

```typescript
import { createServerFn } from "@tanstack/react-start";
import { prisma } from "~/lib/db";
import { NotAuthorized, requireAdmin } from "~/lib/admin-auth";

const listArticles = createServerFn({ method: "GET" }).handler(async () => {
```

### Task 7: `/admin/edit/$slug`, delete old `/edit/$slug`, update 404 link

**IMPORTANT — filename must use the `admin_.` escape prefix, not `admin.`:**
Task 6 discovered that TanStack Router's flat-file convention makes `admin.tsx`
(the `/admin` list route, landed in Task 6) an implicit parent layout for any
sibling file sharing the `admin.` prefix — that file becomes a real child
route (`getParentRoute: () => AdminRoute` in the generated route tree), and

```typescript
// src/routes/admin_.edit.$slug.tsx
import { createFileRoute, useRouter } from "@tanstack/react-router";
import { createServerFn } from "@tanstack/react-start";
import { prisma } from "~/lib/db";
import { NotAuthorized, requireAdmin } from "~/lib/admin-auth";
import { ArticleForm, type ArticleFormValues } from "~/lib/article-form";

const loadArticle = createServerFn({ method: "GET" })
  .inputValidator((slug: string) => slug)
  .handler(async ({ data: slug }) => {
    const a = await prisma.article.findUnique({
      where: { slug },
      select: { slug: true, title: true, summary: true, contentHtml: true },
    });
    return a ?? { slug, title: "", summary: "", contentHtml: "" };
  });

const saveArticle = createServerFn({ method: "POST" })
  .inputValidator((input: ArticleFormValues) => input)
  .handler(async ({ data }) => {
    await prisma.article.upsert({
      where: { slug: data.slug },
      create: {
        slug: data.slug,
        title: data.title,
        summary: data.summary,
        contentHtml: data.contentHtml,
        published: true,
      },
      update: {
        title: data.title,
        summary: data.summary,
        contentHtml: data.contentHtml,
      },
    });
    return { ok: true };
  });

export const Route = createFileRoute("/admin_/edit/$slug")({
  beforeLoad: async ({ location }) => ({
    auth: await requireAdmin(location.href),
  }),
  loader: async ({ params, context }) => {
    if (context.auth.status !== "ok") return null;
    return loadArticle({ data: params.slug });
  },
  component: AdminEditPage,
});

function AdminEditPage() {
  const { auth } = Route.useRouteContext();
  if (auth.status === "unauthorized") return <NotAuthorized email={auth.email} />;

  const initial = Route.useLoaderData()!;
  const router = useRouter();

  return (
    <>
      <h1>Editando: {initial.slug}</h1>
      <ArticleForm
        initial={{
          slug: initial.slug,
          title: initial.title,
          summary: initial.summary ?? "",
          contentHtml: initial.contentHtml,
        }}
        slugEditable={false}
        submitLabel="Guardar"
        onSubmit={async (values) => {
          await saveArticle({ data: values });
          router.navigate({ to: "/article/$slug", params: { slug: values.slug } });
        }}
      />
    </>
  );
}
```
